# 13 度量方式的坑：L2、IP、COSINE 不是同一把尺子

## 这个实验验证什么

同一批向量，L2、IP、COSINE 量出不同的冠军：一条向量可以内积最大（IP 的第一名）、同时距离最远（L2 的倒数第一名）。归一化之后，L2 和 COSINE 又退化成同一把尺子。三把尺子各自保留了什么信息、丢掉了什么信息，这个实验用手造的 2 维向量心算验证，再用 128 维向量看真跑起来的差别。

关联正文：[专栏二·04](../专栏二-Milvus实战/04-数据入库.md)（metric_type 怎么选、归一化后三者排序等价）。

## 实验材料构建

两个脚本：[exp13-gen.py](lab/scripts/exp13-gen.py) 造三组数据、建八个 collection，[exp13-run.py](lab/scripts/exp13-run.py) 做三步对比。

三组数据的设计各有用意：

| 组 | 数据 | 建 collection | 验证什么 |
|----|------|--------------|---------|
| 1 | 手造 2 维向量 5 条：(1,0)、(2,0)、(3,0)、(0,1)、(1,1) | L2 / IP / COSINE 各一个 | 能心算，三把尺子的排序差异一眼看穿 |
| 2 | 128 维随机向量 1000 条 + 10 条 query，**不归一化** | L2 / IP / COSINE 各一个 | 真实维度下三把尺子选出不同的邻居 |
| 3 | 第 2 组归一化后的版本 | L2 / COSINE 各一个 | 归一化后两把尺子是不是合一把 |

第 2 组有个容易漏掉的细节：模长特意拉开到 0.5~2.0 之间均匀采样。如果用纯高斯随机向量，128 维下模长全挤在 11.3 附近（卡方分布在高维集中得厉害），三种度量的排序几乎重合，什么差异都看不出来。想让实验有戏剧性，得先把模长的差异造出来。

```python
# lab/scripts/exp13-gen.py
"""实验 13 材料：三把尺子（L2 / IP / COSINE）量同一批向量。

三组数据：
  1) 手造 2 维向量 5 条，可以心算，建 L2/IP/COSINE 三个 collection；
  2) 128 维随机向量 1000 条 + 10 条 query，模长特意拉开到 0.5~2.0——
     如果用纯高斯向量，128 维下模长全挤在 11.3 附近（卡方分布集中），
     三种度量的排序几乎重合，看不出差异；
  3) 第 2 组归一化后的版本，只建 L2/COSINE 两个 collection。
"""
from pathlib import Path

import numpy as np
from pymilvus import (
    Collection,
    CollectionSchema,
    DataType,
    FieldSchema,
    connections,
    utility,
)

LAB = Path(__file__).resolve().parent.parent
DATA = LAB / "data"

DIM = 128
N = 1000
NQ = 10

VECS_2D = [[1.0, 0.0], [2.0, 0.0], [3.0, 0.0], [0.0, 1.0], [1.0, 1.0]]   # id 1~5


def make_collection(name, dim, metric):
    if utility.has_collection(name):
        utility.drop_collection(name)
    schema = CollectionSchema([
        FieldSchema("id", DataType.INT64, is_primary=True),
        FieldSchema("vector", DataType.FLOAT_VECTOR, dim=dim),
    ])
    col = Collection(name, schema)
    col.create_index(field_name="vector", index_params={
        "index_type": "FLAT", "metric_type": metric, "params": {}})
    col.load()
    return col


def main():
    DATA.mkdir(exist_ok=True)
    connections.connect(host="localhost", port="19530")

    # ---- 第 1 组：2 维心算组 ----
    for metric in ["L2", "IP", "COSINE"]:
        col = make_collection(f"exp13_2d_{metric.lower()}", 2, metric)
        col.insert([list(range(1, 6)), VECS_2D])
        col.flush()
    print("2 维心算组就绪：exp13_2d_l2 / exp13_2d_ip / exp13_2d_cos")

    # ---- 第 2 组：128 维，模长特意拉开 ----
    rng = np.random.default_rng(1313)
    direction = rng.normal(size=(N, DIM))
    direction /= np.linalg.norm(direction, axis=1, keepdims=True)
    length = rng.uniform(0.5, 2.0, size=(N, 1))
    vectors = direction * length

    q_dir = rng.normal(size=(NQ, DIM))
    q_dir /= np.linalg.norm(q_dir, axis=1, keepdims=True)
    q_len = rng.uniform(0.5, 2.0, size=(NQ, 1))
    queries = q_dir * q_len

    for metric in ["L2", "IP", "COSINE"]:
        col = make_collection(f"exp13_128_{metric.lower()}", DIM, metric)
        col.insert([list(range(N)), vectors.tolist()])
        col.flush()
    print("128 维组就绪：exp13_128_l2 / exp13_128_ip / exp13_128_cos（未归一化）")

    # ---- 第 3 组：归一化后的同一批向量（模长全 1）----
    for metric in ["L2", "COSINE"]:
        col = make_collection(f"exp13_128n_{metric.lower()}", DIM, metric)
        col.insert([list(range(N)), direction.tolist()])
        col.flush()
    print("归一化组就绪：exp13_128n_l2 / exp13_128n_cos")

    np.savez(DATA / "exp13.npz", queries=queries)
    print(f"query 已存 {DATA / 'exp13.npz'}，下一步：python scripts/exp13-run.py")


if __name__ == "__main__":
    main()
```

主测量脚本。所有查询都走 Strong 一致性级别的句柄——这个实验要看的差异必须全部归到度量方式头上，"等不等水位"那个变量归实验 14 管，这里把它钉死：

```python
# lab/scripts/exp13-run.py
"""实验 13 主测量：同一批向量，三把尺子，两两对答案。

所有查询都走 Strong 一致性级别的句柄：把"等水位"这个变量钉死，
只看度量方式的差异（那个变量归实验 14 管）。
"""
from pathlib import Path

import numpy as np
from pymilvus import Collection, connections

LAB = Path(__file__).resolve().parent.parent
DATA = LAB / "data"

TOPK = 10
METRICS = ["L2", "IP", "COSINE"]


def strong(name):
    return Collection(name, consistency_level="Strong")


def search_topk(col, metric, q, k):
    res = col.search(data=[q], anns_field="vector",
                     param={"metric_type": metric, "params": {}}, limit=k)
    hits = res[0]
    return list(hits.ids), list(hits.distances)


def main():
    connections.connect(host="localhost", port="19530")

    # ---- 第 1 步：2 维心算组，query = (1, 0) ----
    print("== 第 1 步：2 维心算组，query = (1, 0)，各搜 top-3 ==")
    for metric in METRICS:
        col = strong(f"exp13_2d_{metric.lower()}")
        ids, dists = search_topk(col, metric, [1.0, 0.0], 3)
        print(f"{metric:>6}: " + ", ".join(f"id={i} d={d:.4f}" for i, d in zip(ids, dists)))

    # ---- 第 2 步：128 维未归一化组 ----
    print("\n== 第 2 步：128 维未归一化组，10 条 query 的 top-10 ==")
    queries = np.load(DATA / "exp13.npz")["queries"]
    tops = {}
    for metric in METRICS:
        col = strong(f"exp13_128_{metric.lower()}")
        tops[metric] = [search_topk(col, metric, q.tolist(), TOPK) for q in queries]

    print("query#0 的三份 top-10：")
    for metric in METRICS:
        ids, dists = tops[metric][0]
        print(f"{metric:>6}  ids: {ids}")
        print(f"{'':>6}  dist: {[round(d, 3) for d in dists]}")
    for a, b in [("L2", "IP"), ("L2", "COSINE"), ("IP", "COSINE")]:
        overlap = np.mean([len(set(tops[a][i][0]) & set(tops[b][i][0]))
                           for i in range(len(queries))])
        print(f"top-10 平均重合：{a} vs {b} = {overlap:.1f} / 10")

    # ---- 第 3 步：归一化后，L2 和 COSINE 是不是同一把尺子 ----
    print("\n== 第 3 步：归一化后，同一 query 在 L2 / COSINE 下的 top-10 ==")
    q = queries[0]
    qn = (q / np.linalg.norm(q)).tolist()
    res = {}
    for metric in ["L2", "COSINE"]:
        col = strong(f"exp13_128n_{metric.lower()}")
        res[metric] = search_topk(col, metric, qn, TOPK)
        print(f"{metric:>6}  ids: {res[metric][0]}")
        print(f"{'':>6}  dist: {[round(d, 3) for d in res[metric][1]]}")
    if res["L2"][0] == res["COSINE"][0]:
        print("逐位对比：完全一致——归一化后 L2 和 COSINE 是同一把尺子")
    else:
        print("逐位对比：存在差异！（理论上不该出现，见文档坑点）")


if __name__ == "__main__":
    main()
```

## 实验步骤

**1. 2 维心算组：三把尺子，一个 query，三个答案**

```bash
python scripts/exp13-run.py
```

query 固定用 (1, 0)，在三个 collection 里各搜 top-3。预期你会看到（示意，id 对应关系：(1,0)=id1、(2,0)=id2、(3,0)=id3、(0,1)=id4、(1,1)=id5）：

```text
== 第 1 步：2 维心算组，query = (1, 0)，各搜 top-3 ==
    L2: id=1 d=0.0000, id=2 d=1.0000, id=5 d=1.0000
    IP: id=3 d=3.0000, id=2 d=2.0000, id=1 d=1.0000
COSINE: id=1 d=1.0000, id=2 d=1.0000, id=3 d=1.0000
```

三行各看一眼，先心算对一遍账：

- **L2** 的 top-3 是 (1,0)、(2,0)、(1,1)。注意 (2,0) 和 (1,1) 的距离都是 1，并列第二，谁先谁后是实现细节，两台机器跑出来顺序不同都正常。x 轴上三条向量 (1,0)、(2,0)、(3,0) 的 L2 排序是 1、2、3——按距离近，符合直觉。
- **IP** 的冠军是 (3,0)：内积 (3,0)·(1,0)=3，全库最大。但它恰恰是 L2 眼里**最远**的一条——平方距离 4，全库第一远。两把尺子量出两个冠军，而且 IP 的冠军是 L2 的倒数第一。
- **COSINE** 的前三被 x 轴三条同向向量包揽，距离全是 1.0——只看夹角，模长 1 还是 3 无所谓，(1,0) 和 (3,0) 并列第一。

两个口径问题提前交代，免得你对不上账：其一，并列分数的顺序不定（L2 的 id2/id5、IP 的 id1/id5、COSINE 的 id1/id2/id3 都是并列），你看到的具体顺序可能和上面不同。其二，Milvus 的 L2 返回值是平方距离还是开根号距离，不同资料口径不一——本实验这组数据前 3 名的值两种口径恰好相同；第 4、5 名会露馅，若你看到 2.0 和 4.0 是平方口径，1.414 和 2.0 是开根号口径，以 pymilvus 2.4.4 实际输出为准。排序不受影响（单调对应）。

**2. 128 维组：真跑起来，邻居列表真的不一样**

脚本接着跑 10 条 query 在三个 128 维 collection 的 top-10。预期你会看到（示意，节选；id 具体是哪些不重要，看三行的差别）：

```text
query#0 的三份 top-10：
     L2  ids: [214, 887, 63, 905, 352, 41, 570, 128, 764, 231]
        dist: [1.284, 1.352, 1.397, 1.433, 1.478, 1.502, 1.536, 1.571, 1.602, 1.645]
     IP  ids: [905, 63, 118, 214, 49, 702, 887, 352, 916, 37]
        dist: [0.862, 0.834, 0.817, 0.795, 0.781, 0.774, 0.766, 0.758, 0.743, 0.739]
COSINE  ids: [63, 214, 507, 887, 819, 905, 260, 352, 645, 312]
        dist: [0.372, 0.351, 0.344, 0.331, 0.322, 0.317, 0.309, 0.301, 0.296, 0.288]
top-10 平均重合：L2 vs IP = 3.9 / 10
top-10 平均重合：L2 vs COSINE = 5.6 / 10
top-10 平均重合：IP vs COSINE = 4.8 / 10
```

三件事。第一，邻居列表平均只重合一半不到——同一批数据、同一个 query，换把尺子选出来的 top-10 就是另一批人。第二，三个 distance 字段的大小方向都不统一：L2 越小越好，IP 和 COSINE 越大越好。第三，值域也不统一：L2 下界 0 没上界，IP 完全无界（这组数据模长最长 2，IP 最大也就 4 以内；真实未归一化向量模长可以任意大，IP 的分数跟着任意大），COSINE 死死框在 [-1, 1]。

**3. 归一化验证：L2 和 COSINE 合成一把尺子**

脚本最后把 query 也归一化（除以模长），在归一化 collection 里重跑 top-10。预期你会看到（示意，节选）：

```text
== 第 3 步：归一化后，同一 query 在 L2 / COSINE 下的 top-10 ==
     L2  ids: [63, 214, 507, 887, 819, 905, 260, 352, 645, 312]
        dist: [1.256, 1.298, 1.312, 1.338, 1.356, 1.366, 1.382, 1.398, 1.408, 1.424]
COSINE  ids: [63, 214, 507, 887, 819, 905, 260, 352, 645, 312]
        dist: [0.372, 0.351, 0.344, 0.331, 0.322, 0.317, 0.309, 0.301, 0.296, 0.288]
逐位对比：完全一致——归一化后 L2 和 COSINE 是同一把尺子
```

逐位对比全对上。再心算验证一层：top-1 的 COSINE 是 0.372，L2 是 1.256——2 − 2×0.372 = 1.256，严丝合缝（若你的 L2 是开根号口径，就是 sqrt(1.256)≈1.121，同样单调对应）。这一步是数学恒等式，不是概率现象，跑多少遍都是完全一致。

还有个顺手的观察：第 3 步的 COSINE top-10 和第 2 步的 COSINE top-10 一模一样（都是 [63, 214, 507, …]）——归一化不改夹角，COSINE 本来就把模长除掉了，数据归不归一化它无所谓。而 L2 的 top-10 从第 2 步的"另一批人"变成和 COSINE 完全一致——数据一归一化，L2 的"长度偏见"就没了。

## 现象解释

三把尺子的差别，一句话版本：**IP 量的是投影长度，等于方向相似乘以向量长度；L2 同时看方向差和模长差；COSINE 只看夹角，把模长整个除掉。**

回到 2 维那组数据。IP 为什么选 (3,0) 当冠军：内积 (3,0)·(1,0) = 3×1 + 0×0 = 3——query 方向上的投影长度就是 3，模长直接是放大器。方向对得齐的向量，越长 IP 越喜欢。L2 为什么把它排倒数第一：端点距离平方是 (3-1)² + 0 = 4，模长差全部计价。COSINE 为什么同向三条并列：夹角全是 0 度，模长 1、2、3 统一视而不见。

128 维组重合度不高的原因也在这：L2 的公式里有 ||x||² 一项，罚长向量，top-10 系统性偏向短向量；IP 的公式里 ||x|| 是乘数，捧长向量，top-10 系统性偏向长向量。两把尺子各自筛出的人群，平均只重合一半上下。（想亲眼看这个偏好：在 gen 里加一个标量字段存模长，搜索时用 output_fields 带出来，L2 的 top-10 模长普遍在 0.5 出头，IP 的普遍逼近 2.0。）

归一化之后为什么 L2 和 COSINE 合流：模长全是 1 时，L2 的平方 = ||q||² + ||x||² − 2q·x = 1 + 1 − 2cosθ = 2 − 2cosθ，只剩下夹角一项在动，和 COSINE 单调对应。这正是正文[专栏二·04](../专栏二-Milvus实战/04-数据入库.md)那句"归一化后三者排序完全一致"的出处——本实验第 3 步把它逐位验了。IP 同理也合流（||x||=1 时内积就是 cosθ），第 3 步没建 IP 的 collection，留给你在进一步实验里补。

所以 metric 的选择原则才那么硬：**metric 不是检索参数，是模型的一部分**。模型用什么度量训练、输出什么几何结构，就用什么度量检索。文本 embedding 的信息在方向上、通常已归一化，三者等价，选 COSINE 语义最直白；有些推荐模型的训练目标是最大化内积、输出不归一化，就必须 IP。

## 常见坑点

**换 embedding 模型后沿用旧 metric。** 旧模型输出归一化（模长全 1），三个度量排序等价，随手选了 IP 也看不出问题；换成的新模型不做归一化，IP 的语义就悄悄变了——top-10 开始系统性偏向模长大的向量，检索质量静悄悄地烂掉，不报错，只是结果变差。bge 系列输出通常已归一化，但这事以模型卡为准（待验证），换模型先查卡再动 metric。

**distance 的方向性在业务代码里写反。** L2 越小越好、IP 和 COSINE 越大越好。`if hit.distance > threshold` 这种代码，在 COSINE collection 里是对的，换个 L2 collection 就整个反过来——不报错，只是过滤逻辑全废，属于最隐蔽的那类 bug。写阈值前先确认度量方向。

**阈值跨 metric 迁移。** "相似度大于 0.8"在 COSINE 里是余弦 0.8（约 37 度夹角）；搬到 L2 上 0.8 是平方距离，含义完全变了。换度量方式，阈值必须重标定。

**依赖并列分数的顺序。** 本实验 2 维组里你已经看到并列：COSINE 的前三距离全是 1.0，谁排第一是实现细节。业务上如果用"第一名"做什么逻辑（比如取 top-1 当作唯一答案），并列时行为是不确定的。

## 进一步实验

把 (3,0) 归一化成 (1,0)，重新插一遍 2 维心算组：IP 的冠军换成 (2,0)——内积 2，剩下最长的同向向量。归一化掉最长的，偏好没有消失，只是换了个对象；把 (2,0) 也归一化掉，同向向量长度全变成 1，IP 才和 COSINE 完全一样。一条条归一化，三把尺子的结论逐条趋同——这就是"归一化后三者等价"的动态版。

再补一个：照第 2 组的样子建一个 exp13_128n_ip（归一化数据 + IP），验证三把尺子归一化后完全等价——你手上有 L2 和 COSINE 的结果，差一个 IP 就齐了。

## 最后，总结

IP 量投影，模长是放大器，同向的长向量永远占便宜；L2 方向和模长差都计价；COSINE 把模长除掉只看夹角。三者的 distance 字段方向和值域都不统一，业务代码里的比较和阈值都绑定度量方向。全归一化之后，L2 的平方等于 2−2cos，三把尺子合成一把——所以文本场景"归一化 + COSINE"是默认起点，换模型先查模型卡再定 metric。度量方式跟着模型走，不跟着感觉走。
